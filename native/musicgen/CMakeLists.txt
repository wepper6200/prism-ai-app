cmake_minimum_required(VERSION 3.22)
project(prism_musicgen C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PRISM_MUSICGEN_VULKAN "Build the Vulkan backend" OFF)
set(ACE_ROOT ${CMAKE_CURRENT_LIST_DIR}/../acestep.cpp)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ${PRISM_MUSICGEN_VULKAN} CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_MAX_NAME 128 CACHE STRING "" FORCE)
add_compile_definitions(GGML_MAX_NAME=128)
if(ANDROID)
    set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
endif()
if(MSVC)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
    add_compile_options(/utf-8)
endif()
find_package(Threads REQUIRED)
add_subdirectory(${ACE_ROOT}/ggml ${CMAKE_BINARY_DIR}/ggml)
add_library(prism_ace_json STATIC ${ACE_ROOT}/vendor/yyjson/yyjson.c)
target_include_directories(prism_ace_json SYSTEM PUBLIC ${ACE_ROOT}/vendor/yyjson)
add_library(prism_ace STATIC
    ${ACE_ROOT}/src/request.cpp
    ${ACE_ROOT}/src/model-store.cpp
    ${ACE_ROOT}/src/pipeline-lm.cpp
    ${ACE_ROOT}/src/pipeline-synth.cpp
    ${ACE_ROOT}/src/pipeline-synth-ops.cpp
)
target_include_directories(prism_ace SYSTEM PUBLIC ${ACE_ROOT}/src ${ACE_ROOT})
target_link_libraries(prism_ace PUBLIC prism_ace_json ggml Threads::Threads)
add_executable(prism_musicgen prism_musicgen.cpp)
target_link_libraries(prism_musicgen PRIVATE prism_ace)
if(MSVC)
    target_link_options(prism_musicgen PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_LIST_DIR}/utf8.manifest")
    target_compile_options(prism_musicgen PRIVATE /W4 /WX)
else()
    target_compile_options(prism_musicgen PRIVATE -Wall -Wextra -Werror)
endif()
if(ANDROID)
    set_target_properties(prism_musicgen PROPERTIES PREFIX "lib" SUFFIX ".so")
    target_link_libraries(prism_musicgen PRIVATE log)
else()
    set_target_properties(prism_musicgen PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
    foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(prism_musicgen PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} ${CMAKE_BINARY_DIR}/bin)
    endforeach()
endif()
