cmake_minimum_required(VERSION 3.22)
project(prism_apple C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL ${APPLE} CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
add_subdirectory(../llama.cpp ${CMAKE_BINARY_DIR}/llama)
add_library(prism_apple STATIC prism_native.cpp)
target_link_libraries(prism_apple PUBLIC mtmd llama ggml)
if(APPLE)
    target_sources(prism_apple PRIVATE prism_speech.cpp prism_archive.cpp)
    target_include_directories(prism_apple PRIVATE "${PRISM_APPLE_DEPS}")
endif()
target_include_directories(prism_apple PUBLIC ${CMAKE_CURRENT_LIST_DIR})
if(MSVC)
    target_compile_options(prism_apple PRIVATE /W4 /WX)
else()
    target_compile_options(prism_apple PRIVATE -Wall -Wextra -Werror)
endif()
if(NOT CMAKE_CROSSCOMPILING)
    add_executable(prism_bridge_smoke smoke.cpp)
    target_link_libraries(prism_bridge_smoke PRIVATE prism_apple)
    target_compile_options(prism_bridge_smoke PRIVATE -UNDEBUG)
endif()
